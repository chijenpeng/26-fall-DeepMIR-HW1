\begin{frame}[t]{Why neighbors get confused (1): the low end}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      \textbf{Hypothesis.} Old masters were cut for vinyl, where deep bass makes the stylus jump, so they should carry less sub-bass.

      \vspace{0.4em}
      \textbf{Analysis.} Long-term average spectrum (Welch) of the 1,026 training clips, split into two bands.

      \vspace{0.5em}
      {\scriptsize\setlength{\tabcolsep}{3.5pt}
      \begin{tabular}{@{}lrrrrrr@{}}
        \toprule
        dB re.\ total & 60s & 70s & 80s & 90s & 00s & 10s \\ \midrule
        20--60 Hz  & \alert{$-21.7$} & $-17.4$ & $-16.0$ & $-14.5$ & $-15.0$ & \alert{$-13.1$} \\
        60--120 Hz & $-8.6$ & $-8.0$ & $-8.1$ & $-7.6$ & $-7.6$ & $-7.1$ \\
        \bottomrule
      \end{tabular}}

      \vspace{0.6em}
      \begin{tight}\footnotesize
        \item Sub-bass rises by \alert{8.6 dB} from the 1960s to the 2010s; the 60--120 Hz band barely moves
        \item From the 1990s on the low end is nearly identical, so it cannot separate 1990s / 2000s / 2010s
      \end{tight}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering\includegraphics[width=\linewidth]{lowend_trend.pdf}\\[-0.2em]

    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t]{Why neighbors get confused (2): the loudness war}
  \small
  \textbf{Hypothesis.} From the 2000s, brick-wall limiters make masters louder and flatten peaks.
  \vspace{0.4em}

  \centering
  {\footnotesize
  \begin{tabular}{@{}lrrrrrr@{}}
    \toprule
    Median per decade, training clips & 60s & 70s & 80s & 90s & 00s & 10s \\ \midrule
    RMS level (dBFS)                         & $-14.0$ & $-14.6$ & $-14.5$ & $-13.0$ & \alert{$-11.0$} & \alert{$-10.6$} \\
    Crest factor, peak $-$ RMS (dB)          & 13.9 & 14.5 & 14.5 & 13.0 & \alert{11.0} & \alert{10.6} \\
    Dynamic range, loudest 20\% blocks (dB)  & 10.6 & 11.2 & 11.3 & 10.7 & \alert{8.9}  & \alert{8.7} \\
    Longest run pinned at the peak (samples) & 5 & 5 & 6 & 9 & \alert{22} & \alert{26} \\
    Samples within 0.5 dB of the peak (\%)   & 0.02 & 0.01 & 0.02 & 0.06 & \alert{0.42} & \alert{0.50} \\
    \bottomrule
  \end{tabular}}
  \vspace{0.5em}

  \raggedright
  \begin{tight}\footnotesize
    \item 2000s and 2010s clips are 3 dB louder, have 3 dB less crest, and hold peaks at the ceiling.
          The two are \alert{almost indistinguishable}, matching the 2000s $\leftrightarrow$ 2010s confusion
    \item Production cues only split the data into ``old'' and ``new''; finer distinctions must come from musical content
  \end{tight}
  \vspace{0.5em}
  {\scriptsize\color{gray}\textbf{Caveat.} The clips are 24 kHz mono, decoded from lossy Opus (YouTube). Resampling, downmixing and lossy coding
  reshape waveform peaks, so these numbers are trends (ANOVA $p < 10^{-30}$), not exact properties of the original masters.
  The two sample-level rows are the most affected.\par}
\end{frame}

\begin{frame}[t]{Task 1 as year regression: why it fails}
  \small
  Both rows use the final Task 1 features (Whisper-tiny layer 1 + MERT-v1-330M layers 5, 6, 14); only the predictor differs.\\
  Number of validation clips each formulation assigns to each decade:
  \vspace{0.5em}

  \centering
  {\footnotesize\setlength{\tabcolsep}{7pt}
  \begin{tabular}{@{}lrrrrrr@{}}
    \toprule
    Clips predicted as & 1960s & 1970s & 1980s & 1990s & 2000s & 2010s \\ \midrule

    Logistic regression (final model) & 27 & 20 & 19 & 21 & 18 & 27 \\
    Ridge regression     & \alert{4} & 22 & 35 & 38 & 28 & \alert{5} \\
    \bottomrule
  \end{tabular}}
  \vspace{0.8em}

  \raggedright
  \begin{itemize}
    \item \alert{Regression is much worse} ($-20.5$ points top-1): predictions collapse toward the middle, only 4 clips are called 1960s and 5 are called 2010s
    \item Tuning does not fix it: with a ridge penalty from $10^2$ to $10^6$ the best top-1 is 32\%; stronger regularization worsens the collapse
    \item The classifier already learns the order by itself (see the t-SNE slide), so nothing is gained from imposing it
  \end{itemize}

\section{Ablation}

\begin{frame}[t,label=ablation]{Ablation (1): how the features are chosen}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      {\small\textbf{Design idea}}
      \begin{tight}\footnotesize
        \item Two frozen encoders trained for different things: \textbf{Whisper} (speech recognition) for the voice and the \alert{language}, \textbf{MERT} (music understanding) for the \alert{music}: timbre, instruments, production
        \item Which encoder and which layer fit each task is measured, not assumed
        \item \textbf{Task 1}: we also test our hand-crafted production features (low end, loudness, dynamics). They do \alert{not} improve the model (next slide)
        \item \textbf{Task 2}: we add the 7-d language ID of Whisper-small as an explicit language cue
      \end{tight}
    \end{column}
    \begin{column}{0.5\textwidth}
      {\small\textbf{Selection procedure (both tasks)}}
      \begin{enumerate}\footnotesize\setlength{\itemsep}{2pt}
        \item \textbf{Whisper}: sweep every layer of all five sizes (tiny to large-v3) and keep the best single layer
        \item \textbf{MERT}: for 95M and for 330M, read three layer sets off the layer sweep:
          \begin{itemize}\footnotesize
            \item the best layer
            \item the three best layers
            \item one layer per third of the depth: the best shallow, middle and deep layer
          \end{itemize}
        \item \textbf{Combine} every MERT set with the Whisper layer and compare validation top-1
      \end{enumerate}
    \end{column}
  \end{columns}
  \vspace{0.2em}
  {\scriptsize\color{gray}Every number: logistic regression on mean + std pooled features, fitted on train, scored on validation, $C \in \{3{\times}10^{-4}, 10^{-3}, 3{\times}10^{-3}, 10^{-2}\}$ chosen on validation.\par}
\end{frame}
