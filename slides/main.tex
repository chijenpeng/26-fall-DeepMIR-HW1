  \vspace{0.3em}
  \archlegend{41,478}
  \vspace{0.1em}

  \raggedright\small
  \begin{tight}\footnotesize
    \item \textbf{Whisper-tiny, layer 1}: shallow layers work best for decade; accuracy drops with depth
    \item \textbf{MERT-v1-330M, layers 5, 6, 14}: best three of a 25-layer sweep; adding Whisper-tiny gives +2.3 points
    \item \textbf{mean + std pooling}: one $d$-dim vector per frame $\rightarrow$ mean and std \textbf{over the frames} $\rightarrow$ $2 \times d$ values per layer

    \item Only the classifier is trained, with strong L2 (1,026 clips, 6,912 features)
  \end{tight}

\begin{frame}[t]{Why neighbors get confused (1): the low end}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      \textbf{Hypothesis.} Old masters were cut for vinyl, where deep bass makes the stylus jump, so they should carry less sub-bass.

      \vspace{0.4em}
      \textbf{Analysis.} Long-term average spectrum (Welch) of the 1,026 training clips, split into two bands.

      \vspace{0.5em}
      {\scriptsize\setlength{\tabcolsep}{3.5pt}
      \begin{tabular}{@{}lrrrrrr@{}}
        \toprule
        dB re.\ total & 60s & 70s & 80s & 90s & 00s & 10s \\ \midrule
        20--60 Hz  & \alert{$-21.7$} & $-17.4$ & $-16.0$ & $-14.5$ & $-15.0$ & \alert{$-13.1$} \\
        60--120 Hz & $-8.6$ & $-8.0$ & $-8.1$ & $-7.6$ & $-7.6$ & $-7.1$ \\
        \bottomrule
      \end{tabular}}

      \vspace{0.6em}
      \begin{tight}\footnotesize
        \item Sub-bass rises by \alert{8.6 dB} from the 1960s to the 2010s; the 60--120 Hz band barely moves
        \item From the 1990s on the low end is nearly identical, so it cannot separate 1990s / 2000s / 2010s
      \end{tight}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering\includegraphics[width=\linewidth]{lowend_trend.pdf}\\[-0.2em]
      {\scriptsize\color{gray}Median and inter-quartile range, training split.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t]{Why neighbors get confused (2): the loudness war}
  \small
  \textbf{Hypothesis.} From the 2000s, brick-wall limiters make masters louder and flatten peaks.
  \vspace{0.4em}

  \centering
  {\footnotesize
  \begin{tabular}{@{}lrrrrrr@{}}
    \toprule
    Median per decade, training clips & 60s & 70s & 80s & 90s & 00s & 10s \\ \midrule
    RMS level (dBFS)                         & $-14.0$ & $-14.6$ & $-14.5$ & $-13.0$ & \alert{$-11.0$} & \alert{$-10.6$} \\
    Crest factor, peak $-$ RMS (dB)          & 13.9 & 14.5 & 14.5 & 13.0 & \alert{11.0} & \alert{10.6} \\
    Dynamic range, loudest 20\% blocks (dB)  & 10.6 & 11.2 & 11.3 & 10.7 & \alert{8.9}  & \alert{8.7} \\
    Longest run pinned at the peak (samples) & 5 & 5 & 6 & 9 & \alert{22} & \alert{26} \\
    Samples within 0.5 dB of the peak (\%)   & 0.02 & 0.01 & 0.02 & 0.06 & \alert{0.42} & \alert{0.50} \\
    \bottomrule
  \end{tabular}}
  \vspace{0.5em}

  \raggedright
  \begin{tight}\footnotesize
    \item 2000s and 2010s clips are 3 dB louder, have 3 dB less crest, and hold peaks at the ceiling.
          The two are \alert{almost indistinguishable}, matching the 2000s $\leftrightarrow$ 2010s confusion
    \item Production cues only split the data into ``old'' and ``new''; finer distinctions must come from musical content
  \end{tight}
  \vspace{0.5em}
  {\scriptsize\color{gray}\textbf{Caveat.} The clips are 24 kHz mono, decoded from lossy Opus (YouTube). Resampling, downmixing and lossy coding
  reshape waveform peaks, so these numbers are trends (ANOVA $p < 10^{-30}$), not exact properties of the original masters.
  The two sample-level rows are the most affected.\par}
\end{frame}

  \raggedright\small
  \begin{tight}\footnotesize
    \item \textbf{Whisper-large-v3, layers 23, 30}: deepest layers work best, the opposite of Task 1; this fits language being the main cue
    \item \textbf{MERT-v1-95M, layer 7}: +2 points on top of Whisper (57.8\% $\rightarrow$ 59.8\%); a music encoder adds what a speech encoder misses
    \item \textbf{Whisper-small language ID}: an explicit 7-d posterior; a small gain (top-3 89.2\% $\rightarrow$ 90.2\%) because deep Whisper layers already encode language
    \item Pooling as in Task 1: 1,500 Whisper frames per clip; 374 MERT frames per 5 s chunk
  \end{tight}

\begin{frame}[t]{Why English-language markets get confused: language}
  \small
  The dataset has \textbf{no language label}. We estimate it with Whisper-small \textbf{language identification}
  on the 798 training clips, on the mixture and on Demucs-separated vocal stems.
  \vspace{0.5em}

  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      {\footnotesize\setlength{\tabcolsep}{4.5pt}
      \begin{tabular}{@{}llrrr@{}}
        \toprule
        & & \multicolumn{1}{c}{Home} & \multicolumn{2}{c}{Detected as English} \\ \cmidrule(l){4-5}
        Market & Home language & \multicolumn{1}{c}{mixture} & mixture & vocals \\ \midrule
        US      & English    & 92\%  & --   & -- \\
        UK      & English    & 100\% & --   & -- \\
        Brazil  & Portuguese & 87\%  & 11\% & 12\% \\
        Spain   & Spanish    & 57\%  & 39\% & 41\% \\
        Germany & German     & 17\%  & \alert{83\%} & \alert{82\%} \\
        Italy   & Italian    & 41\%  & \alert{55\%} & \alert{56\%} \\
        \bottomrule
      \end{tabular}}

      \vspace{0.4em}
      {\scriptsize\color{gray}Share of training clips whose most likely language (among en, pt, es, de, it) is the home language or English.\par}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{tight}\footnotesize
        \item Most German releases and over half of the Italian ones are \alert{sung in English}, so their vocals carry no market cue
        \item Vocal stems give the same answer: the accompaniment is not what misleads the detector
        \item Language separates Brazil and Spain. The English-language group has to be split by genre and production style, a much weaker cue
        \item A release market is where the record was issued, not the artist's nationality or language
      \end{tight}
    \end{column}
  \end{columns}
\end{frame}
